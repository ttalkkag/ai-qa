호스트 도구 또는 사본 안의 실행 파일만 사용하세요. 셸 명령 문자열·인증정보·원본 절대 경로는 허용하지 않습니다.

[
  {
    "command_id": "unit",
    "display_name": "단위 테스트",
    "kind": "TEST",
    "parser_id": "pytest-summary",
    "parser_version": 1,
    "argv": [".venv/bin/python", "-m", "pytest", "-q"],
    "cwd": ".",
    "timeout_sec": 300,
    "required": true,
    "generated_paths": [".aiqa-output"]
  }
]
